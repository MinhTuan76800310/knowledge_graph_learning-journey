% header.tex — LaTeX preamble customizations for the Knowledge Graph book.
% Engine: lualatex (Unicode + fontspec). Vietnamese rendered via system fonts.

% --- Fonts (cross-platform fonts with full Vietnamese coverage) ---
\usepackage{fontspec}
% Symbols like ★ / ⚠ may be absent from Times New Roman; fall back
% to Segoe UI for missing glyphs instead of printing tofu boxes.
\directlua{
  luaotfload.add_fallback("symfallback", {
    "Cambria Math:mode=node;",
    "Segoe UI:mode=node;",
  })
}
\setmainfont{Times New Roman}[RawFeature={fallback=symfallback}]
\setsansfont{Segoe UI}
% Consolas lacks precomposed Vietnamese glyphs (ồ, ề, ỉ, ...);
% fall back to Segoe UI for the missing ones so code comments render.
\setmonofont{Consolas}[Scale=0.85, RawFeature={fallback=symfallback}]

% Symbols like ★ / ⚠ may be absent from Times New Roman; the Lua filter
% (longtable-filter.lua) replaces them with \ding{} / \ensuremath{} equivalents
% before they reach the LaTeX engine, avoiding font subsetting issues entirely.
\usepackage{amssymb}
\usepackage{pifont}
\usepackage{unicode-math}
\setmathfont{Cambria Math}

% --- Vietnamese language support ---
% Pandoc already loads babel with \babelprovide[main,import]{vietnamese}
% because metadata.yaml sets lang: vi; loading babel again here would cause
% an option clash.

% --- Tables: use longtable so wide/tall tables auto-break across pages ---
\usepackage{longtable}
\usepackage{booktabs}

% --- Code blocks: allow line breaking so they never overflow the margin ---
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\}}

% --- Figures: cap width and height so images never clip or overflow ---
\makeatletter
\def\maxwidth{%
  \ifdim\Gin@nat@width>\linewidth
    \linewidth
  \else
    \Gin@nat@width
  \fi
}
\def\maxheight{%
  \ifdim\Gin@nat@height>0.82\textheight
    0.82\textheight
  \else
    \Gin@nat@height
  \fi
}
\makeatother
\let\oldincludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \oldincludegraphics[#1,width=\maxwidth,height=\maxheight,keepaspectratio]{#2}}

% --- Reduce overfull-box noise ---
\hbadness=10000
\hfuzz=2pt
